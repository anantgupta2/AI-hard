[theorem] [Similarity]\label{thm:similarity} Let \(A\) be any unital complex \(C^*\)-algebra, let \(H\) be any complex Hilbert space, and let \(\pi:A\to\mathcal B(H)\) be a bounded, complex-linear, unital algebra homomorphism. There exists an operator \(S\in\mathcal B(H)\), invertible with bounded inverse, such that \[ \rho(a)=S\pi(a)S^{-1}\qquad(a\in A) \] is a \(*\)-homomorphism. In particular, \(\rho(a^*)=\rho(a)^*\) for every \(a\in A\).
[theorem] [Uniform commutator estimate]\label{thm:universal} There is an absolute constant \(C<\infty\) such that, for every complex Hilbert space \(K\), every unital von Neumann algebra \(P\subset\mathcal B(K)\), every \(Y\in\mathcal B(K)\), every integer \(h\ge1\), and every \(X\in M_h(P)\), \begin{equation}\label{eq:universal} \|[Y^{(h)},X]\|\le C\,g_P(Y)\,\|X\|. \end{equation} Here \(M_h(P)\) acts naturally on \(K^h\), and \([T,U]=TU-UT\).
[corollary] [Universal hyperreflexivity]\label{cor:hyperreflexivity} Let \(H\) be any complex Hilbert space and let \(M\subset\mathcal B(H)\) be a unital von Neumann algebra. For \(T\in\mathcal B(H)\), set \[ \alpha_M(T)=\sup\{\|(1-e)Te\|:e\in M',\ e=e^*=e^2\}. \] Then, with the absolute constant \(C\) from Theorem~\ref{thm:universal}, \begin{equation}\label{eq:universal-hyperreflexivity} \dist(T,M)\le 2C\,\alpha_M(T)\qquad(T\in\mathcal B(H)). \end{equation} Thus every such \(M\) is hyperreflexive with constant at most \(2C\), independently of the algebra, its representation, and the Hilbert space. No optimality of this constant is asserted.
[theorem] [Kirchberg's similarity--derivation equivalence] \label{thm:kirchberg} For a unital \(C^*\)-algebra \(A\), the following properties are equivalent. \begin{enumerate} \item Every bounded unital homomorphism \(\pi:A\to\mathcal B(H)\), on any Hilbert space \(H\), is similar by a bounded invertible operator to a \(*\)-homomorphism. \item For every faithful unital \(*\)-representation \(\sigma:A\to\mathcal B(K)\), on any Hilbert space \(K\), every bounded linear map \(\Delta:A\to\mathcal B(K)\) satisfying \[ \Delta(ab)=\Delta(a)\sigma(b)+\sigma(a)\Delta(b) \] has the form \(\Delta(a)=V\sigma(a)-\sigma(a)V\) for some \(V\in\mathcal B(K)\). \end{enumerate}
[theorem] [Free-product corner estimate]\label{thm:free-corner} Let $\mathcal L=D*A_0$ be as above, let $t=1/N$ for an integer $N\geq2$, and suppose that $G\in\mathcal B(L^2(\mathcal L))$ commutes with $L_D$. Suppose that $\epsilon\geq0$ and \begin{equation}\label{eq:free-corner-hypothesis} \|[G,L_x]\|\leq\epsilon\|x\| \qquad (x\in e\mathcal L e, \ e\in D\text{ a projection with }\tau(e)=t). \end{equation} If $s=s^*=s^{-1}\in A_0$ and $\tau(s)=0$, then \begin{equation}\label{eq:free-corner-conclusion} \left\|P_{L^2(A_0)}[G,L_s]\big|_{L^2(A_0)}\right\| \leq c_1\epsilon, \qquad c_1=6\pi^2(1+4c_0). \end{equation}
[proposition] [Finite standard multiplicities] \label{prop:finite-copies} Let \(M\) be a separable-predual \(\mathrm{II}_1\) factor, and let \(M\) act on \(H_r=L^2(M)^r\) by common left multiplication, where \(r\geq1\) is finite. For every \(Y\in\mathcal B(H_r)\), every finite \(h\geq1\), and every \(X\in M_h(M)\), \[ \norm{[Y^{(h)},X]} \leq (3c_1+2)\,g_M(Y)\,\norm{X}. \] Here \(X\) denotes its entrywise action on \(H_r^h\); the constant is independent of \(M,r,h\).
[proposition] \label{prop:factor-representations} Let $P\subset\B(K)$ be a unital factor acting on a separable Hilbert space. For every $Y\in\B(K)$, every $h\ge1$, and every $X\in M_h(P)$, \[ \norm{[Y^{(h)},X]} \le C_{\mathrm{fac}}\,g_P(Y)\,\norm{X}. \]
[proposition] \label{prop:separable-estimate} Let $P\subset\B(K)$ be any unital von Neumann algebra on a separable Hilbert space. Then \[ \norm{[Y^{(h)},X]}\le C\,g_P(Y)\,\norm{X} \qquad(Y\in\B(K),\ X\in M_h(P),\ h\ge1). \]
[proposition] \label{prop:all-derivations} Let $A$ be a unital $C^*$-algebra and $\sigma:A\to\B(K)$ a unital $*$-representation. Every bounded $\sigma$-derivation $\Delta:A\to\B(K)$ satisfies \[ \norm{\Delta}_{\mathrm{cb}}\le C\norm{\Delta} \] and is implemented by a bounded operator on $K$.

namespace Kadison
variable {K : Type u} [NormedAddCommGroup K] [InnerProductSpace ℂ K]
  [CompleteSpace K]
def commutatorSeminorm (P : VonNeumannAlgebra K) (Y : K →L[ℂ] K) : ℝ :=
  sSup {r : ℝ | ∃ a : K →L[ℂ] K, a ∈ P ∧ ‖a‖ ≤ 1 ∧ r = ‖Y * a - a * Y‖}
def matrixOperator (h : ℕ) (X : Matrix (Fin h) (Fin h) (K →L[ℂ] K)) :
    (PiLp 2 fun _ : Fin h => K) →L[ℂ] (PiLp 2 fun _ : Fin h => K) :=
  (PiLp.continuousLinearEquiv 2 ℂ (fun _ : Fin h => K)).symm.toContinuousLinearMap.comp
    (ContinuousLinearMap.pi (fun i => ∑ j : Fin h,
      (X i j).comp (PiLp.proj 2 (fun _ : Fin h => K) j)))
def diagonalAmplification (h : ℕ) (Y : K →L[ℂ] K) :
    (PiLp 2 fun _ : Fin h => K) →L[ℂ] (PiLp 2 fun _ : Fin h => K) :=
  (PiLp.continuousLinearEquiv 2 ℂ (fun _ : Fin h => K)).symm.toContinuousLinearMap.comp
    (ContinuousLinearMap.pi (fun i => Y.comp (PiLp.proj 2 (fun _ : Fin h => K) i)))
theorem uniform_commutator :
    ∃ C : ℝ, 0 ≤ C ∧ ∀ (K : Type u) [NormedAddCommGroup K] [InnerProductSpace ℂ K]
      [CompleteSpace K] (P : VonNeumannAlgebra K) (Y : K →L[ℂ] K)
      (h : ℕ), 1 ≤ h → ∀ (X : Matrix (Fin h) (Fin h) (K →L[ℂ] K)),
      (∀ i j, X i j ∈ P) →
      ‖diagonalAmplification h Y * matrixOperator h X -
        matrixOperator h X * diagonalAmplification h Y‖ ≤
        C * commutatorSeminorm P Y * ‖matrixOperator h X‖ := by
  sorry
